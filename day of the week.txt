class Solution {
    public String dayOfTheWeek(int day, int month, int year) {
        String[] daysOfWeek = {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};
        int[] daysInMonths = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

        int totalDays = 0;

        // Count days from years 1971 to year - 1
        for (int y = 1971; y < year; y++) {
            totalDays += isLeapYear(y) ? 366 : 365;
        }

        // Adjust February if current year is a leap year
        if (isLeapYear(year)) {
            daysInMonths[1] = 29;
        }

        // Count days from previous months
        for (int m = 0; m < month - 1; m++) {
            totalDays += daysInMonths[m];
        }

        // Add remaining days
        totalDays += day;

        // Jan 1, 1971 was Friday (index 5 in [Sun, Mon, Tue, Wed, Thu, Fri, Sat])
        return daysOfWeek[(totalDays + 4) % 7];
    }

    private boolean isLeapYear(int y) {
        return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
    }
}
